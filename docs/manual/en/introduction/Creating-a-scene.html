using System.Collections; using System.Collections.Generic; using UnityEngine; using UnityEngine.UI; // For UI elements public class CharacterAbilities : MonoBehaviour { public float health = 100f; public float damage = 10f; public float swingSpeed = 10f; public float venomStrikePower = 20f; public float focusBar = 100f; public float focusGainRate = 10f; private bool isSwinging = false; private bool isVenomStriking = false; private string[] abilitySlots; private Dictionary abilityCooldowns = new Dictionary(); private float abilityCooldownTime = 5f; public Text healthText; public Text focusText; public enum SuitType { Regular, Symbiote, AntiVenom, Miles, Venom } public SuitType currentSuit = SuitType.Regular; void Start() { abilitySlots = new string[4]; // Initialize ability slots UpdateAbilities(); UpdateUI(); } void Update() { for (int i = 0; i < 4; i++) { if (Input.GetKeyDown(KeyCode.Alpha1 + i)) UseAbility(i); } if (Input.GetKeyDown(KeyCode.E)) UseFocus(); } void UpdateAbilities() { if (abilitySlots == null) abilitySlots = new string[4]; switch (currentSuit) { case SuitType.Regular: abilitySlots = new string[] { "Web Barrage", "Spider Reflex", "Web Slam", "Ultimate Swing" }; break; case SuitType.Symbiote: abilitySlots = new string[] { "Rage Punch", "Tendrils Whip", "Symbiote Surge", "Devour" }; break; case SuitType.AntiVenom: abilitySlots = new string[] { "Purge Blast", "Cleansing Strike", "Healing Touch", "Symbiote Surge" }; break; case SuitType.Miles: abilitySlots = new string[] { "Venom Punch", "Venom Dash", "Venom Blast", "Camouflage" }; break; case SuitType.Venom: abilitySlots = new string[] { "Tendril Slam", "Ravenous Bite", "Symbiote Surge", "Shadow Leap" }; break; } } void UseAbility(int slot) { if (slot < 0 || slot >= abilitySlots.Length || string.IsNullOrEmpty(abilitySlots[slot])) { Debug.LogWarning("Invalid ability slot selected!"); return; } if (abilityCooldowns.ContainsKey(slot) && Time.time < abilityCooldowns[slot]) { Debug.LogWarning("Ability is on cooldown!"); return; } abilityCooldowns[slot] = Time.time + abilityCooldownTime; Debug.Log(abilitySlots[slot] + " activated!"); } public void TakeDamage(float damageAmount) { health = Mathf.Max(health - damageAmount, 0); Debug.Log($"Character took {damageAmount} damage. Health: {health}"); UpdateUI(); if (health <= 0) { Die(); } } public void GainFocus(float amount) { focusBar = Mathf.Clamp(focusBar + amount, 0, 100f); Debug.Log($"Focus increased by {amount}. Current Focus: {focusBar}"); UpdateUI(); } void UseFocus() { if (focusBar >= 50f) { focusBar -= 50f; Heal(20f); Debug.Log("Focus used to heal!"); } else { Debug.Log("Not enough focus!"); } UpdateUI(); } void Heal(float healAmount) { health = Mathf.Clamp(health + healAmount, 0, 100f); Debug.Log($"Healed {healAmount} health. Current health: {health}"); UpdateUI(); } void Die() { Debug.Log("Character is dead."); } void UpdateUI() { if (healthText != null) healthText.text = "Health: " + health; if (focusText != null) focusText.text = "Focus: " + focusBar; } } // Boss Class public class Boss : MonoBehaviour { public string bossName; public float maxHealth = 1000f; public float currentHealth; public float damage = 50f; public float transitionHealthThresholds[] = { 75f, 50f, 25f }; private int currentPhase = 0; private float attackCooldown = 0f; public GameObject deathEffectPrefab; public Text bossHealthText; private bool isSymbioteShedding = false; public enum BossType { Venom, Scream, PeterSymbiote } public BossType bossType; private void Start() { currentHealth = maxHealth; UpdateUI(); } private void Update() { if (Time.time >= attackCooldown) { PerformAttack(); } if (currentHealth <= transitionHealthThresholds[currentPhase] && currentPhase < 3) { TransitionToNextPhase(); } if (isSymbioteShedding) { HandleSymbioteShedding(); } UpdateUI(); } void UpdateUI() { if (bossHealthText != null) bossHealthText.text = $"{bossName} Health: {currentHealth}/{maxHealth}"; } public void TakeDamage(float damageAmount) { currentHealth -= damageAmount; if (currentHealth <= 0) { Die(); } } void PerformAttack() { switch (bossType) { case BossType.Venom: VenomAttack(); break; case BossType.Scream: ScreamAttack(); break; case BossType.PeterSymbiote: PeterSymbioteAttack(); break; } attackCooldown = Time.time + 3f; } void VenomAttack() { switch (currentPhase) { case 0: Debug.Log("Venom uses basic web attack!"); break; case 1: Debug.Log("Venom uses aggressive symbiote tendrils!"); break; case 2: Debug.Log("Venom summons symbiote minions!"); break; case 3: Debug.Log("Venom enters frenzy mode, using a devastating shockwave!"); break; } } void ScreamAttack() { switch (currentPhase) { case 0: Debug.Log("Scream uses sonic scream!"); break; case 1: Debug.Log("Scream uses poison gas burst!"); break; case 2: Debug.Log("Scream summons symbiote creatures to fight!"); break; case 3: Debug.Log("Scream enrages and releases a scream shockwave!"); break; } } void PeterSymbioteAttack() { switch (currentPhase) { case 0: Debug.Log("Peter Symbiote uses web barrage!"); break; case 1: Debug.Log("Peter Symbiote uses symbiote tendrils!"); break; case 2: Debug.Log("Peter Symbiote summons black symbiote clones!"); break; case 3: if (!isSymbioteShedding) { Debug.Log("Peter is struggling with the symbiote... something is happening!"); isSymbioteShedding = true; } break; } } void TransitionToNextPhase() { currentPhase++; Debug.Log($"{bossName} is transitioning to Phase {currentPhase + 1}!"); ApplyPhaseEffects(); } void ApplyPhaseEffects() { switch (currentPhase) { case 1: maxHealth *= 0.75f; break; case 2: maxHealth *= 0.5f; break; case 3: maxHealth *= 0.25f; break; } currentHealth = maxHealth; } void HandleSymbioteShedding() { Debug.Log("Peter is ripping the symbiote off!"); currentPhase = 4; isSymbioteShedding = false; maxHealth += 200f; damage = 30f; Debug.Log("Peter is back to his normal self, ready to fight!"); ApplyPhaseEffects(); } void Die() { Debug.Log($"{bossName} is dead!"); if (deathEffectPrefab != null) { Instantiate(deathEffectPrefab, transform.position, Quaternion.identity); } Destroy(gameObject); } public bool IsAlive() { return currentHealth > 0; } } // Boss Manager for spawning and managing bosses public class BossManager : MonoBehaviour { public GameObject venomBossPrefab; public GameObject screamBossPrefab; public GameObject peterSymbioteBossPrefab; public Transform spawnLocation; private Boss currentBoss; public void StartBossFight(Boss.BossType bossType) { GameObject bossObject = null; switch (bossType) { case Boss.BossType.Venom: bossObject = Instantiate(venomBossPrefab, spawnLocation.position, Quaternion.identity); break; case Boss.BossType.Scream: bossObject = Instantiate(screamBossPrefab, spawnLocation.position, Quaternion.identity); break; case Boss.BossType.PeterSymbiote: bossObject = Instantiate(peterSymbioteBossPrefab, spawnLocation.position, Quaternion.identity); break; } if (bossObject != null) { currentBoss = bossObject.GetComponent(); Debug.Log($"Boss fight started with {currentBoss.bossName}"); } } }
